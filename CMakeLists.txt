cmake_minimum_required(VERSION 3.20)
project(CShift LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# LLVM (tested with LLVM 22). Set LLVM_DIR to <llvm>/lib/cmake/llvm if it is not found automatically.
find_package(LLVM REQUIRED CONFIG)
message(STATUS "Found LLVM ${LLVM_PACKAGE_VERSION} (${LLVM_DIR})")

include_directories(${LLVM_INCLUDE_DIRS})
separate_arguments(LLVM_DEFINITIONS_LIST NATIVE_COMMAND ${LLVM_DEFINITIONS})
add_definitions(${LLVM_DEFINITIONS_LIST})

# Embed the standard library (stdlib/*.csh, written in CShift) into the compiler as byte arrays.
file(GLOB STDLIB_FILES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/stdlib/*.csh)
list(SORT STDLIB_FILES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${STDLIB_FILES})
set(STDLIB_DATA_CPP ${CMAKE_CURRENT_BINARY_DIR}/StdlibData.cpp)
set(STDLIB_CONTENT "// Generated by CMake from stdlib/*.csh. Do not edit.\n#include \"StdlibData.h\"\n\n")
set(STDLIB_TABLE "")
set(STDLIB_INDEX 0)
foreach(STDLIB_FILE ${STDLIB_FILES})
    get_filename_component(STDLIB_NAME ${STDLIB_FILE} NAME)
    file(READ ${STDLIB_FILE} STDLIB_HEX HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," STDLIB_BYTES "${STDLIB_HEX}")
    string(APPEND STDLIB_CONTENT "static const unsigned char kData${STDLIB_INDEX}[] = {${STDLIB_BYTES}0x00};\n")
    string(APPEND STDLIB_TABLE "    {\"${STDLIB_NAME}\", kData${STDLIB_INDEX}, sizeof(kData${STDLIB_INDEX}) - 1},\n")
    math(EXPR STDLIB_INDEX "${STDLIB_INDEX}+1")
endforeach()
string(APPEND STDLIB_CONTENT "\nconst StdlibFile kStdlibFiles[] = {\n${STDLIB_TABLE}};\nconst unsigned kStdlibFileCount = ${STDLIB_INDEX};\n")
set(STDLIB_OLD "")
if(EXISTS ${STDLIB_DATA_CPP})
    file(READ ${STDLIB_DATA_CPP} STDLIB_OLD)
endif()
if(NOT STDLIB_OLD STREQUAL STDLIB_CONTENT)
    file(WRITE ${STDLIB_DATA_CPP} "${STDLIB_CONTENT}")
endif()

add_executable(cshiftc
    ${STDLIB_DATA_CPP}
    compiler/src/main.cpp
    compiler/src/Types.cpp
    compiler/src/Lexer.cpp
    compiler/src/Parser.cpp
    compiler/src/CodeGen.cpp
    compiler/src/CodeGenRuntime.cpp
    compiler/src/CodeGenExpr.cpp
    compiler/src/CodeGenCall.cpp
    compiler/src/CodeGenStmt.cpp
)

target_include_directories(cshiftc PRIVATE compiler/src)

# Link the complete LLVM (shared library if LLVM was built that way, otherwise all static components).
if(LLVM_LINK_LLVM_DYLIB)
    target_link_libraries(cshiftc PRIVATE LLVM)
else()
    llvm_map_components_to_libnames(LLVM_LIBS all)
    target_link_libraries(cshiftc PRIVATE ${LLVM_LIBS})
endif()

if(MSVC)
    target_compile_options(cshiftc PRIVATE /W3 /permissive- /utf-8)
else()
    target_compile_options(cshiftc PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
