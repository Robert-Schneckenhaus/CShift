cmake_minimum_required(VERSION 3.20)
# C is enabled because LLVM's CMake package runs C-based feature checks (e.g. for libedit) while it is loaded.
project(CShift LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# LLVM (tested with LLVM 22). Set LLVM_DIR to <llvm>/lib/cmake/llvm if it is not found automatically.
find_package(LLVM REQUIRED CONFIG)
message(STATUS "Found LLVM ${LLVM_PACKAGE_VERSION} (${LLVM_DIR})")

include_directories(${LLVM_INCLUDE_DIRS})
separate_arguments(LLVM_DEFINITIONS_LIST NATIVE_COMMAND ${LLVM_DEFINITIONS})
add_definitions(${LLVM_DEFINITIONS_LIST})

# Embed the standard library (stdlib/*.csh, written in CShift) into the compiler as byte arrays.
file(GLOB STDLIB_FILES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/stdlib/*.csh)
list(SORT STDLIB_FILES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${STDLIB_FILES})
set(STDLIB_DATA_CPP ${CMAKE_CURRENT_BINARY_DIR}/StdlibData.cpp)
set(STDLIB_CONTENT "// Generated by CMake from stdlib/*.csh. Do not edit.\n#include \"StdlibData.h\"\n\n")
set(STDLIB_TABLE "")
set(STDLIB_INDEX 0)
foreach(STDLIB_FILE ${STDLIB_FILES})
    get_filename_component(STDLIB_NAME ${STDLIB_FILE} NAME)
    file(READ ${STDLIB_FILE} STDLIB_HEX HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," STDLIB_BYTES "${STDLIB_HEX}")
    string(APPEND STDLIB_CONTENT "static const unsigned char kData${STDLIB_INDEX}[] = {${STDLIB_BYTES}0x00};\n")
    string(APPEND STDLIB_TABLE "    {\"${STDLIB_NAME}\", kData${STDLIB_INDEX}, sizeof(kData${STDLIB_INDEX}) - 1},\n")
    math(EXPR STDLIB_INDEX "${STDLIB_INDEX}+1")
endforeach()
string(APPEND STDLIB_CONTENT "\nconst StdlibFile kStdlibFiles[] = {\n${STDLIB_TABLE}};\nconst unsigned kStdlibFileCount = ${STDLIB_INDEX};\n")
set(STDLIB_OLD "")
if(EXISTS ${STDLIB_DATA_CPP})
    file(READ ${STDLIB_DATA_CPP} STDLIB_OLD)
endif()
if(NOT STDLIB_OLD STREQUAL STDLIB_CONTENT)
    file(WRITE ${STDLIB_DATA_CPP} "${STDLIB_CONTENT}")
endif()

add_executable(cshiftc
    ${STDLIB_DATA_CPP}
    compiler/src/main.cpp
    compiler/src/Project.cpp
    compiler/src/FfiImport.cpp
    compiler/src/FfiGenerator.cpp
    compiler/src/Types.cpp
    compiler/src/Lexer.cpp
    compiler/src/Parser.cpp
    compiler/src/CodeGen.cpp
    compiler/src/CodeGenRuntime.cpp
    compiler/src/CodeGenExpr.cpp
    compiler/src/CodeGenCall.cpp
    compiler/src/CodeGenStmt.cpp
)

target_include_directories(cshiftc PRIVATE compiler/src)

# Version shown by 'cshiftc --version' (the release workflow passes -DCSHIFT_VERSION=1.00).
set(CSHIFT_VERSION "dev" CACHE STRING "Version of cshiftc")
target_compile_definitions(cshiftc PRIVATE CSHIFT_VERSION="${CSHIFT_VERSION}")

# C header import (using X from "header.h") needs libclang's C API header; libclang itself is loaded at run time.
find_path(CLANG_C_INCLUDE_DIR clang-c/Index.h HINTS ${LLVM_INCLUDE_DIRS})
if(CLANG_C_INCLUDE_DIR)
    message(STATUS "libclang header found: FFI header import enabled")
    target_include_directories(cshiftc PRIVATE ${CLANG_C_INCLUDE_DIR})
    target_compile_definitions(cshiftc PRIVATE CSHIFT_HAVE_LIBCLANG)
else()
    message(WARNING "clang-c/Index.h not found: FFI header import will be disabled (.ffi files still work)")
endif()

# Standalone executable: LLVM and the C++ runtime are linked statically, so cshiftc.exe runs from any shell
# without the MSYS2 DLLs in PATH (only 'clang' is still needed at run time for the final link step).
option(CSHIFT_STATIC "Link LLVM and the C++ runtime statically (standalone cshiftc)" OFF)

# Link the complete LLVM (shared library if LLVM was built that way, otherwise all static components).
if(CSHIFT_STATIC)
    execute_process(COMMAND ${LLVM_TOOLS_BINARY_DIR}/llvm-config --link-static --libs all
                    OUTPUT_VARIABLE LLVM_STATIC_LIBS OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(COMMAND ${LLVM_TOOLS_BINARY_DIR}/llvm-config --link-static --system-libs
                    OUTPUT_VARIABLE LLVM_STATIC_SYSTEM_LIBS OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(COMMAND ${LLVM_TOOLS_BINARY_DIR}/llvm-config --libdir
                    OUTPUT_VARIABLE LLVM_STATIC_LIB_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
    string(REPLACE "-lzstd.dll" "-lzstd" LLVM_STATIC_SYSTEM_LIBS "${LLVM_STATIC_SYSTEM_LIBS}") # no DLL dependency
    separate_arguments(LLVM_STATIC_LIBS NATIVE_COMMAND "${LLVM_STATIC_LIBS}")
    separate_arguments(LLVM_STATIC_SYSTEM_LIBS NATIVE_COMMAND "${LLVM_STATIC_SYSTEM_LIBS}")
    # LLVM may have been built with the Z3 solver (Ubuntu's packages are); cshiftc does not use it.
    list(FILTER LLVM_STATIC_SYSTEM_LIBS EXCLUDE REGEX "z3")
    # llvm-config prints -lLLVM... without a search path; the libraries are not in the default path on Linux
    # (/usr/lib/llvm-22/lib).
    target_link_directories(cshiftc PRIVATE ${LLVM_STATIC_LIB_DIR})
    target_link_libraries(cshiftc PRIVATE ${LLVM_STATIC_LIBS} ${LLVM_STATIC_SYSTEM_LIBS})
    if(WIN32)
        target_link_options(cshiftc PRIVATE -static -s)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        # Not fully static: loading libclang at run time needs the dynamic loader.
        target_link_options(cshiftc PRIVATE -static-libstdc++ -static-libgcc -s)
    endif()
elseif(LLVM_LINK_LLVM_DYLIB)
    target_link_libraries(cshiftc PRIVATE LLVM)
else()
    llvm_map_components_to_libnames(LLVM_LIBS all)
    target_link_libraries(cshiftc PRIVATE ${LLVM_LIBS})
endif()

if(MSVC)
    target_compile_options(cshiftc PRIVATE /W3 /permissive- /utf-8)
else()
    target_compile_options(cshiftc PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
