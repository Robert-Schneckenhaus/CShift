// Fast trigonometry from tables, for games and demos (raycasting, rotations, fake 3D) on computers without an FPU:
// angles are integers with 1024 steps per full circle, the results are fixed point numbers. No floating point is
// used, so it is fast on a 68000 as well.
//
//     int angle = FastTrig.FromDegrees(30);            // 85 (1024 steps = 360 degrees)
//     int x = (radius * FastTrig.Cos(angle)) >> 14;    // Sin and Cos: 1.14 fixed point, 16384 = 1.0
//     int y = (radius * FastTrig.Sin(angle)) >> 14;
//     int dir = FastTrig.Atan2(dy, dx);                // the angle of a vector, 0..1023
//
// Every int is an angle: it is taken modulo 1024 (with a mask), so negative angles and sums past a full turn work.
// Tan, Cot, Sec (1/cos) and Csc (1/sin) are 16.16 fixed point (65536 = 1.0); at their poles (tan at 90 degrees, ...)
// they return 2147483647 or -2147483647. Sec and Csc give the step lengths of a raycaster's grid walk (DDA).
// The tables are generated by stdlib/tools/fasttrig.py.

namespace FastTrig;

const int Steps = 1024;     // a full circle
const int Mask = 1023;      // angle & Mask: the angle in 0..1023
const int Quarter = 256;    // 90 degrees
const int Half = 512;       // 180 degrees
const int SinOne = 16384;   // 1.0 in the results of Sin and Cos
const int FixedOne = 65536; // 1.0 in the results of Tan, Cot, Sec and Csc

// sin and cos in 1.14 fixed point: -16384..16384
int16 Sin(int angle) { return _SinTable[angle & Mask]; }
int16 Cos(int angle) { unchecked { return _SinTable[(angle + Quarter) & Mask]; } }

// tan, cot, 1/cos and 1/sin in 16.16 fixed point
int Tan(int angle) { return _TanTable[angle & (Half - 1)]; }
int Cot(int angle) { unchecked { return _TanTable[(Quarter - angle) & (Half - 1)]; } }
int Sec(int angle) { return _SecTable[angle & Mask]; }
int Csc(int angle) { unchecked { return _SecTable[(angle - Quarter) & Mask]; } }

// The angle of the vector (x, y) in steps, 0..1023 (0 for the zero vector); exact to about half a step.
int Atan2(int y, int x)
{
    if (x == 0 && y == 0)
        return 0;
    int64 ax = x < 0 ? -(int64)x : (int64)x;
    int64 ay = y < 0 ? -(int64)y : (int64)y;
    while (ax > 32767 || ay > 32767)
    {
        ax = ax >> 1;
        ay = ay >> 1;
    }
    int sx = (int)ax;
    int sy = (int)ay;
    int a;
    if (sx >= sy)
        a = _AtanTable[(sy * 256 + sx / 2) / sx];
    else
        a = Quarter - _AtanTable[(sx * 256 + sy / 2) / sy];
    if (x < 0)
        a = Half - a;
    if (y < 0)
        a = (Steps - a) & Mask;
    return a;
}

// degrees to angle steps (rounded) and back (0..359)
int FromDegrees(int degrees)
{
    int64 scaled = (int64)degrees * 128;
    return (int)(scaled >= 0 ? (scaled + 22) / 45 : (scaled - 22) / 45);
}

int ToDegrees(int angle)
{
    return ((angle & Mask) * 45 + 64) / 128 % 360;
}
